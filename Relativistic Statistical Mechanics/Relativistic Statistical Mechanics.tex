\section{The invariance of entropy}
\begin{theorem}
The entropy $S=k_B\ln\Omega$ for a moving inertial observer, $K$,with respect to the container of an ideal gas ($N$ equal particles of same mass, elastic collisions between the particles and the walls, and non-interacting particles) is equal to the entropy $S_0=k_B\ln\Omega_0$ for a rest observer with the container, $K_0$. 

Where $\Omega=\Omega(E,V,N)$ is the number of states function for $K$ depending on internal energy $E$, volume $V$, and number of particles $N$. Analogously, $\Omega_0=\Omega_0(E_0,V_0,N_0)$ is the number of states function for $K_0$.
\end{theorem}

\textit{Proof:}

By definition:
\be 
\Omega=\frac{1}{N!}\frac{1}{h^{3N}}\underbrace{\int\limits_\mathscr{V} d^3r_1\cdots d^3r_N}_{\Omega_v}\,\underbrace{\int\limits_R d^3p_1 \cdots d^3 p_N}_{\Omega_p}\,,
\ee
where $h$ is the planck constant, and $\vec{r}_i$ and $\vec{p}_i$ are the ith position and momentum for $K$. Also, $\mathscr{V}$ is the volume space, $R$ is the momentum space defined as:
\be 
R = \left\{ (\vec{p}_1,...,\vec{p}_N) \mid \sum\limits_{i=1}^{N} c\sqrt{m^2c^2 + |\vec{p}_i|^2} = E \right\}\,.
\ee

By the non-interacting condition, the position $\vec{r}_i$ of each particle is independent one of the other. And therefore we can separate $\Omega_v$ as a product of integrals:
\be 
	\Omega_v = \left(\int\limits_V d^3 r_1\right)\cdots \left(\int\limits_V d^3 r_N\right) = \left(\int\limits_V d^3 r_1\right)^N = V^N\,.
\ee

\begin{lemma}
The volume $V_0$ for the observer $K_0$ changed to $V_0/\gamma=V$ by a Lorentz transformation corresponding to a Lorentz Boost in one direction. $\gamma$ is the Lorentz factor.
\end{lemma}

By the previous lemma:
\be 
\Omega_v = \frac{V^N}{\gamma^N} = \frac{\Omega_{v0}}{\gamma^N}\,. \label{eq:omega_v}
\ee

By the other side, chosen an appropriate coordinate axis we can say that the Lorentz Boost occur in the $x-$axis. According to \cite{Landau} the transformation laws for momentum in this case are:
\begin{eqnarray}
	p_x &=& \gamma\left(g_x + \frac{v}{c^2}E_0\right) \,, \\
	p_y &=& g_y \,, \\
	p_z &=& g_z \,,
\end{eqnarray}
where $\vec{p}$ is the momentum of one particle for observer $K$, and $\vec{g}$ is the momentum of the same particle but for $K_0$. Besides, $v$ is the relative speed between the observers, $\gamma=1/\sqrt{1-v^2/c^2}$, $c$ is the speed of light, and $E_0=c\sqrt{m^2c^2+g^2}=mc^2\sqrt{1+\frac{g^2}{m^2c^2}}$ is the relativistic energy of the particle for $K_0$.

We want to implement a change of variables from $p_i$ to $g_i$ in the momentum integrals. The Jacobian of the transformation is:
\be 
	J(\vec{g}) = \frac{\partial p_{x^i}}{\partial g_{x^i}} =  \begin{vmatrix}
     \partd{p_x}{g_x} & \partd{p_x}{g_y} & \partd{p_x}{g_z}\\ 
     0 & 1 & 0\\
     0 & 0 & 1  
\end{vmatrix} = \partd{p_x}{g_x} = \partd{ }{g_x}\left[ \gamma \left(g_x + \frac{v}{c^2}mc^2\sqrt{1+\frac{g^2}{m^2c^2}} \right)\right]\,.
\ee

Knowing that $g^2=g_x\,^2+g_y\,^2+g_z\,^2$:
\be 
	J(\vec{g}) = \gamma \left( 1 + \frac{v}{c^2}mc^2 \frac{g_x/(m^2c^2)}{\sqrt{1+\frac{g^2}{m^2c^2}}} \right) = \gamma \left( 1 + \frac{v}{c^2} \frac{g_x/m}{\sqrt{1+\frac{g^2}{m^2c^2}}} \right)\,.
\ee

Then, by implement this coordinate transformation for all the variables of the $N$ particles:
\be 
	\Omega_p = \int\limits_R d^3p_1 \cdots d^3p_N = \int\limits_R \gamma^N \prod\limits_{i=1}^{N}\left( 1 + \frac{v}{c^2} \frac{g_{x_i}/m}{\sqrt{1+\frac{g_i\,^2}{m^2c^2}}} \right) d^3 g_1 \cdots d^3g_N\,. \label{eq:omega_p}
\ee

By using equations \ref{eq:omega_v} and \ref{eq:omega_p} we have that the number of states function is:
\be 
	\Omega = \frac{1}{N!}\frac{1}{h^{3N}}\frac{V^N}{\gamma^N} \int\limits_R \gamma^N \prod\limits_{i=1}^{N}\left( 1 + \frac{v}{c^2} \frac{g_{x_i}/m}{\sqrt{1+\frac{g_i\,^2}{m^2c^2}}} \right) d^3 g_1 \cdots d^3g_N\,,
\ee
\be 
	\Omega = \frac{1}{N!}\frac{1}{h^{3N}} V^N \int\limits_R \gamma^N \prod\limits_{i=1}^{N}\left( 1 + \frac{v}{c^2} \frac{g_{x_i}/m}{\sqrt{1+\frac{g_i\,^2}{m^2c^2}}} \right) d^3 g_1 \cdots d^3g_N\,
\ee
