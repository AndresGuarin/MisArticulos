\section{The invariance of entropy}
\begin{theorem}
The entropy $S=k_B\ln\Omega$ for a moving inertial observer $K$ with respect to the container of an ideal gas ($N$ equal particles of same mass, elastic collisions between particles and walls, and non-interacting particles) is equal to the entropy $S_0=k_B\ln\Omega_0$ for a rest observer $K_0$ with the container. 

Here $\Omega=\Omega(E,V,N)$ is the number of states for $K$, $E$ is the internal energy $E$, $V$ is the volume, and $N$ is the number of particles. Analogously, $\Omega_0=\Omega_0(E_0,V_0,N_0)$ is the number of states for $K_0$, with is corresponding internal energy, volume and number of particles.
\end{theorem}

\textit{Proof:}

By definition
\be 
\Omega=\frac{1}{N!}\frac{1}{h^{3N}}\underbrace{\int\limits_\mathscr{V} d^3r_1\cdots d^3r_N}_{\Omega_v}\,\underbrace{\int\limits_\mathcal{R} d^3p_1 \cdots d^3 p_N}_{\Omega_p}\,,
\ee
where $h$ is the planck constant, and $\vec{r}_i$ and $\vec{p}_i$ are the ith position and momentum for $K$. Also, $\mathscr{V}$ is the volume space, $\mathcal{R}$ is the momentum space defined as
\be 
\mathcal{R} = \left\{ (\vec{p}_1,\dots,\vec{p}_N) \mid \vec{p}_i\cdot \vec{p}_i < 0  \text{  y  } \sum\limits_{i=1}^N \vec{p}_i = \vec{\mathscr{P}}\right\}
\ee


\be 
\mathcal{R} = \left\{ (\vec{p}_1,...,\vec{p}_N) \mid \sum\limits_{i=1}^{N} c\sqrt{m^2c^2 + |\vec{p}_i|^2} = E \right\}\,.
\ee

By the non-interacting condition, the position $\mathbf{r}_i$ of each particle is independent one of the other. And therefore we can separate $\Omega_v$ as a product of integrals
\be 
	\Omega_v = \left(\int\limits_V d^3 r_1\right)\cdots \left(\int\limits_V d^3 r_N\right) = \left(\int\limits_V d^3 r_1\right)^N = V^N\,.
\ee

\begin{lemma}
The volume $V_0$ for the observer $K_0$ changes to $V_0/\gamma=V$ by a Lorentz transformation corresponding to a Lorentz Boost in one direction. $\gamma$ is the Lorentz factor.
\end{lemma}

By the previous lemma 
\be 
\Omega_v = \frac{V^N}{\gamma^N} = \frac{\Omega_{v_0}}{\gamma^N}\,. \label{eq:omega_v}
\ee

By the other side, chosen an appropriate coordinate axis we can say that the Lorentz Boost occur in the $x-$axis. According to \cite{Landau} the transformation laws for momentum in this case are 
\begin{eqnarray}
	p_x &=& \gamma\left(g_x + \frac{v}{c^2}E_0\right) \,, \\
	p_y &=& g_y \,, \\
	p_z &=& g_z \,,
\end{eqnarray}
where $\mathbf{p}$ is the momentum of one particle for observer $K$, and $\mathbf{g}$ is the momentum of the same particle but for $K_0$. Besides, $v$ is the relative speed between the observers, $\gamma=1/\sqrt{1-v^2/c^2}$, $c$ is the speed of light, and $E_0=c\sqrt{m^2c^2+g^2}=mc^2\sqrt{1+\frac{g^2}{m^2c^2}}$ is the relativistic energy of the particle for $K_0$.

We want to implement a change of variables from $p_i$ to $g_i$ in the momentum integrals. The Jacobian of the transformation is 
\be 
	J(\mathbf{g}) = \frac{\partial p_{x^i}}{\partial g_{x^i}} =  \begin{vmatrix}
     \partd{p_x}{g_x} & \partd{p_x}{g_y} & \partd{p_x}{g_z}\\ 
     0 & 1 & 0\\
     0 & 0 & 1  
\end{vmatrix} = \partd{p_x}{g_x} = \partd{ }{g_x}\left[ \gamma \left(g_x + \frac{v}{c^2}mc^2\sqrt{1+\frac{g^2}{m^2c^2}} \right)\right]\,.
\ee

Knowing that $g^2=g_x\,^2+g_y\,^2+g_z\,^2$ 
\be 
	J(\mathbf{g}) = \gamma \left( 1 + \frac{v}{c^2}mc^2 \frac{g_x/(m^2c^2)}{\sqrt{1+\frac{g^2}{m^2c^2}}} \right) = \gamma \left( 1 + \frac{v}{c} \frac{g_x/(mc)}{\sqrt{1+\frac{g^2}{m^2c^2}}} \right)\,.
\ee

Then, by implement this coordinate transformation for all the variables of the $N$ particles 
\be 
	\Omega_p = \int\limits_\mathcal{R} d^3p_1 \cdots d^3p_N = \int\limits_\mathcal{R} \gamma^N \prod\limits_{i=1}^{N}\left( 1 + \frac{v}{c} \frac{g_{x_i}/(mc)}{\sqrt{1+\frac{g_i\,^2}{m^2c^2}}} \right) d^3 g_1 \cdots d^3g_N\,. \label{eq:omega_p}
\ee

One can check that
\begin{eqnarray}
	\prod\limits_{i=1}^{N}(1+M_i) &=& 1+M_1+M_2+\cdots+M_N+ \notag  \\  
								 & & +M_1 M_2 +M_1M_3+\cdots+M_1M_N+\cdots M_{N-1}M_n+ \notag \\
								 & & +M_1M_2M_3 +\cdots M_1 M_{N-1}M_N + \cdots M_{N-2}M_{N-1}M_N + \cdots\,, \notag \\
								 &=& 1 + \sum\limits_{k=1}^{N} \bigsum \limits_{1\leq i_1 < i_2 < \cdots < i_k\leq N} M_{i_1}\cdots M_{i_k}\,,
\end{eqnarray}

where $\sum_{1\leq i_1 < i_2 < \cdots < i_k\leq N}$ stands for the sum over all integers $i_1,\dots,i_k$ that satisfies $1\leq i_1 < i_2 < \cdots < i_k\leq N$. Therefore we have

\be 
	\Omega_p = \int\limits_\mathcal{R} \gamma^N \left( 1 + \sum\limits_{k=1}^{N} \bigsum\limits_{1\leq i_1 < i_2 < \cdots < i_k\leq N} \frac{v^k}{c^k} \frac{g_{x_{i_1}}/(mc)}{\sqrt{1+\frac{g_{i_i}\,^2}{m^2c^2}}} \cdots \frac{g_{x_{i_k}}/(mc)}{\sqrt{1+\frac{g_{i_k}\,^2}{m^2c^2}}} \right) d^3 g_1 \cdots d^3g_N\,.
\ee

As integration is a linear operation

\be 
	\Omega_p =  \gamma^N \int\limits_\mathcal{R} d^3 g_1 \cdots d^3g_N  + \gamma^N \sum\limits_{k=1}^{N} \bigsum\limits_{1\leq i_1 < i_2 < \cdots < i_k\leq N} \frac{v^k}{c^k} \int\limits_\mathcal{R} \frac{g_{x_{i_1}}/(mc)}{\sqrt{1+\frac{g_{i_i}\,^2}{m^2c^2}}} \cdots \frac{g_{x_{i_k}}/(mc)}{\sqrt{1+\frac{g_{i_k}\,^2}{m^2c^2}}} d^3 g_1 \cdots d^3g_N\,.
\ee

While integrating the second term we will find that if a point $(\mathbf{g}_1,\dots,\mathbf{g}_k)$ with corresponding $x-$components $(g_{x_i},\dots, g_{x_k})$ is in $\mathcal{R}$, then $(-\mathbf{g}_1,\dots,-\mathbf{g}_k)$ also will be in $\mathcal{R}$ since $|\mathbf{g}_i|=|-\mathbf{g}_i|$. And so, the volume element centered at $(\mathbf{g}_1,\dots,\mathbf{g}_k)$ is canceled out with the one at $(-\mathbf{g}_1,\dots,-\mathbf{g}_k)$, this is
\be 
	\int\limits_\mathcal{R} \frac{g_{x_{i_1}}/(mc)}{\sqrt{1+\frac{g_{i_i}\,^2}{m^2c^2}}} \cdots \frac{g_{x_{i_k}}/(mc)}{\sqrt{1+\frac{g_{i_k}\,^2}{m^2c^2}}} d^3 g_1 \cdots d^3g_N = 0\,.
\ee

Then only the first term remains
\be 
\	\Omega_p =  \gamma^N \int\limits_\mathcal{R} d^3 g_1 \cdots d^3g_N\,.
\ee

This integral is exactly the definition of $\Omega_p$ but for $K_0$. Hence $\Omega_p=\gamma^N \Omega_{p_0}$, and so $\Omega=\Omega_0$. This ends the proof.
